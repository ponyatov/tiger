# `string`: Строки

```ocaml
let greeting = "Hello, world!"
let empty_string = ""
let multiline = "Line 1\nLine 2"
```

- строковые операции (через модуль String):

```ocaml
String.length "hello"    (* длина строки → 5 *)
String.make 3 'a'        (* создание строки → "aaa" *)
"Hello, " ^ "OCaml!"     (* конкатенация → "Hello, OCaml!" *)
```

- конкатенация (объединение строк)

```ocaml
let user = "Алексей"
let message = "Привет, " ^ user ^ "! Добро пожаловать в систему."
(* message = "Привет, Алексей! Добро пожаловать в систему." *)

let a = 3
let b = 4
let result = "Результат: " ^ string_of_int a ^ " + " ^ string_of_int b ^ " = " ^ string_of_int (a + b)
```

- специальные символы (escape)
	- `\n` перенос строки

```ocaml
let single_line = "Первая строка\nВторая строка\nТретья строка"

(* Вывод:
Первая строка
Вторая строка
Третья строка
*)
```

- многострочные строки

```ocaml
(* обычные кавычки могут быть и многострочными *)
let single = "Первая строка
Вторая строка
Третья строка
"

(* строковый литерат с отключенными спецсимволами и кавычками *)
let multiline = {|
Это многострочная строка.
Она может содержать "кавычки" без экранирования.
Переменные не интерполируются: имя = %s
|}

(* Вложенные кавычки без экранирования *)
let dialog = {|
Он сказал: "Привет!"
А она ответила: 'Здравствуй!'
"Использовать 'одинарные' внутри "двойных"" - это нормально.
|}
```
